#version 430 core

layout(local_size_x = 4, local_size_y = 4) in;
layout(rgba32f, binding = 0) uniform image2D framebufferTex;

struct Triangle
{
    int materialIndex;
    vec3 a;
    vec3 b;
    vec3 c;
    vec3 normal;
};

layout(std430, binding = 1) readonly buffer b_trianglesSSBO
{
    Triangle modelData[];
};

struct Node
{
    vec3 minBound;
    int leftIndex;
    vec3 maxBound;
    int rightIndex;

    int firstTriangle;
    int triangleCount;
};

layout(std430, binding = 2) readonly buffer b_bvhTriangleIndex
{
    int bvhTriangleIndexes[];
};

layout(std430, binding = 3) readonly buffer b_bvhNodes
{
    Node bvhNodes[];
};

struct Camera
{
    vec3 position;
    vec3 forward;
    vec3 right;
    vec3 up;
    float fov;
};

uniform Camera u_camera;

uniform vec2 resolution;

struct Light
{
    vec3 color;
    vec3 position;
};

#define MAX_LIGHTS 10
uniform Light lights[MAX_LIGHTS];
uniform int numLights;

struct Ray 
{
    vec3 origin;
    vec3 direction;
};

struct RaycastResult 
{
    bool hit;
    float distance;
    vec3 position;
    vec3 normal;
    int triangleIndex;
    int materialIndex;
};

RaycastResult triangleIntersect(Ray _ray, Triangle _tri, int _index)
{
    RaycastResult result;
    result.hit = false;
    result.triangleIndex = _index;

    float l_ep = 0.0001;

    vec3 v0 = _tri.a;
    vec3 v1 = _tri.b; 
    vec3 v2 = _tri.c;

    vec3 edge1 = v1 - v0;
    vec3 edge2 = v2 - v0;
    vec3 ray_cross_edge2 = cross(_ray.direction, edge2);
    float det = dot(edge1, ray_cross_edge2);

    // Only check triangles facing the _ray
    if(det < l_ep)
    {
        return result;
    }

    float inv_det = 1.0 / det;
    vec3 s = _ray.origin - v0;
    float u = inv_det * dot(s, ray_cross_edge2);

    if(u < 0.0 || u > 1.0)
    {
        return result;
    }

    vec3 s_cross_e1 = cross(s, edge1);
    float v = inv_det * dot(_ray.direction, s_cross_e1);

    if(v < 0.0 || u + v > 1.0)
    {
        return result;
    }

    float t = inv_det * dot(edge2, s_cross_e1);

    if(t > l_ep)
    {
        result.hit = true;
        result.position = _ray.origin + _ray.direction * t;
        result.distance = t;
        result.normal = _tri.normal;
        return result;
    }

    return result;
}

bool IntersectAABB(Ray _ray, vec3 _minBound, vec3 _maxBound)
{
    float tmin = 0.0f;          // Start with the minimum distance (can be -FLT_MAX for entire ray)
    float tmax = 10000000.0f;    // Maximum allowable distance for the ray (segment length or ∞)

    // Iterate over each axis (x, y, z)
    for (int i = 0; i < 3; i++)
    {
        // If the ray is parallel to the slab (AABB plane pair)
        if (abs(_ray.direction[i]) < 0.0001f)
        {
            // If the origin is outside the slab, there's no intersection
            if (_ray.origin[i] < _minBound[i] || _ray.origin[i] > _maxBound[i])
                return false;
        }
        else
        {
            // Compute the intersection t-values for the near and far planes of the slab
            float ood = 1.0f / _ray.direction[i];
            float t1 = (_minBound[i] - _ray.origin[i]) * ood;
            float t2 = (_maxBound[i] - _ray.origin[i]) * ood;

            // Ensure t1 is the intersection with the near plane, and t2 with the far plane
            if (t1 > t2)
            {
                float temp = t1; // Swap t1 and t2
                t1 = t2;
                t2 = temp;
            }

            // Update tmin and tmax to compute the intersection interval
            tmin = max(tmin, t1);
            tmax = min(tmax, t2);

            // If the interval becomes invalid, there is no intersection
            if (tmin > tmax)
            {
                return false;
            }
        }
    }

    return true;
}

RaycastResult rayNodeIntersect(inout RaycastResult _result, Ray _ray, int _nodeIndex)
{
    Node node = bvhNodes[_nodeIndex];
    if(!IntersectAABB(_ray, node.minBound, node.maxBound))
    {
        return _result;
    }

    // Leaf node
    if(node.triangleCount > 0)
    {
        for(int i = 0; i < node.triangleCount; i++)
        {
            RaycastResult newResult = triangleIntersect(_ray, modelData[bvhTriangleIndexes[node.firstTriangle + i]], bvhTriangleIndexes[node.firstTriangle + i]);
            if(newResult.hit && newResult.distance < _result.distance)
            {
                _result = newResult;
            }
        }
    }
    else
    {
        rayNodeIntersect(_result, _ray, bvhNodes[_nodeIndex].leftIndex);
        rayNodeIntersect(_result, _ray, bvhNodes[_nodeIndex].rightIndex);
    }

    return _result;
}

bool isInShadow(vec3 lightPos, RaycastResult raycastHit)
{
    Ray shadowRay;
    shadowRay.origin = raycastHit.position + raycastHit.normal * 0.001;
    shadowRay.direction = normalize(lightPos - raycastHit.position);
    
    float lightDistance = length(lightPos - raycastHit.position);
    
    rayNodeIntersect(raycastHit, shadowRay, 0);
    
    return false;
}


vec3 getObjectColor(RaycastResult hit)
{
    return modelData[hit.triangleIndex].normal;
}

vec3 calculateShading(RaycastResult hit, Ray _ray)
{
    if (!hit.hit)
    {
        return vec3(0.1, 0.1, 0.3); // Background color
    }
    
    vec3 objectColor = getObjectColor(hit);
    vec3 lighting = vec3(0.0);

    for(int i = 0; i < numLights; i++)
    {
        // Check if point is in shadow for this light
        if (isInShadow(lights[i].position, hit))
        {
            continue;
        }
        
        // Calculate direction from hit point to light
        vec3 lightDir = normalize(lights[i].position - hit.position);

        // Diffuse component
        float diffuse = max(dot(hit.normal, lightDir), 0.0);
        vec3 diffuseColor = lights[i].color * objectColor * diffuse;

        // Specular component
        vec3 viewDir = normalize(u_camera.position - hit.position);
        vec3 reflectDir = reflect(-lightDir, hit.normal);
        float specular = pow(max(dot(viewDir, reflectDir), 0.0), 32.0);
        vec3 specularColor = lights[i].color * specular;

        // Add attenuation based on distance
        float distance = length(lights[i].position - hit.position);
        float attenuation = 1.0 / (1.0 + 0.1 * distance + 0.01 * distance * distance);
        
        lighting += (diffuseColor + specularColor) * attenuation;
    };
    
    // Add ambient lighting (always present, even in shadows)
    vec3 ambient = objectColor * 0.1;
    
    return ambient + lighting;
}

void main() 
{
    vec2 uv = vec2(((float(gl_GlobalInvocationID.x) / gl_NumWorkGroups.x) - 0.5) * 2.0,
                   ((float(gl_GlobalInvocationID.y) / gl_NumWorkGroups.y) - 0.5) * 2.0);

    float aspect = gl_NumWorkGroups.x / gl_NumWorkGroups.y;
    vec3 rayDir = normalize(u_camera.forward + 
                           uv.x * u_camera.right * aspect + 
                           uv.y * u_camera.up);
    
    Ray _ray;
    _ray.origin = u_camera.position;
    _ray.direction = rayDir;
    
    RaycastResult closestHit;
    closestHit.hit = false;
    closestHit.distance = 1e10;

    rayNodeIntersect(closestHit, _ray, 0);

    vec3 color = calculateShading(closestHit, _ray);
    ivec2 position = ivec2(gl_GlobalInvocationID.xy);
    imageStore(framebufferTex, position, vec4(color, 1.0));
}
