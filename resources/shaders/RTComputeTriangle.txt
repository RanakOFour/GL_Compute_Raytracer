struct RaycastResult 
{
    bool hit;
    float distance;
    vec3 position;
    vec3 normal;
    int triangleIndex;
};

RaycastResult triangleIntersect(Ray _ray, Triangle _tri, int _index)
{
    RaycastResult result;
    result.hit = false;
    result.triangleIndex = _index;

    float l_ep = 0.0001;

    vec3 v0 = _tri.a;
    vec3 v1 = _tri.b; 
    vec3 v2 = _tri.c;

    vec3 edge1 = v1 - v0;
    vec3 edge2 = v2 - v0;
    vec3 ray_cross_edge2 = cross(_ray.direction, edge2);
    float det = dot(edge1, ray_cross_edge2);

    if(det > -l_ep && det < l_ep)
    {
        return result;
    }

    float inv_det = 1.0 / det;
    vec3 s = _ray.origin - v0;
    float u = inv_det * dot(s, ray_cross_edge2);

    if(u < 0.0 || u > 1.0)
    {
        return result;
    }

    vec3 s_cross_e1 = cross(s, edge1);
    float v = inv_det * dot(_ray.direction, s_cross_e1);

    if(v < 0.0 || u + v > 1.0)
    {
        return result;
    }

    float t = inv_det * dot(edge2, s_cross_e1);

    if(t > l_ep)
    {
        result.hit = true;
        result.position = _ray.origin + _ray.direction * t;
        result.distance = t;
        result.normal = _tri.normal;
        return result;
    }

    return result;
}

bool IntersectAABB(inout Ray _ray, vec3 _minBound, vec3 _maxBound)
{
    float tmin = 0.0f;
    float tmax = 10000000.0f;

    // Iterate over each axis (x, y, z)
    for (int i = 0; i < 3; i++)
    {
        if (abs(_ray.direction[i]) < 0.0001f)
        {
            // If the origin is outside the slab, there's no intersection
            if (_ray.origin[i] < _minBound[i] || _ray.origin[i] > _maxBound[i])
                return false;
        }
        else
        {
            // Compute the intersection t-values for the near and far planes of the slab
            float ood = 1.0f / _ray.direction[i];
            float t1 = (_minBound[i] - _ray.origin[i]) * ood;
            float t2 = (_maxBound[i] - _ray.origin[i]) * ood;

            // Ensure t1 is the intersection with the near plane, and t2 with the far plane
            if (t1 > t2)
            {
                float temp = t1; // Swap t1 and t2
                t1 = t2;
                t2 = temp;
            }

            // Update tmin and tmax to compute the intersection interval
            tmin = max(tmin, t1);
            tmax = min(tmax, t2);

            // If the interval becomes invalid, there is no intersection
            if (tmin > tmax)
            {
                return false;
            }
        }
    }

    return true;
}


RaycastResult TraverseBVH(inout RaycastResult _result, inout Ray _ray)
{
    int indexStack[64];
    int top = 0;
    indexStack[top] = 0;
    int hits = 0;

    Node node = bvhNodes[indexStack[top]];
    top++;

    do
    {
        if(IntersectAABB(_ray, node.minBound, node.maxBound))
        {
            if(node.triangleCount > 0)
            {
                for(int i = 0; i < node.triangleCount; i++)
                {
                    RaycastResult newResult = triangleIntersect(_ray, modelData[bvhTriangleIndexes[node.leftFirst + i]], bvhTriangleIndexes[node.leftFirst + i]);
                    if(newResult.hit && newResult.distance < _result.distance)
                    {
                        _result = newResult;
                    }
                }
            }
            else
            {
                indexStack[top] = node.leftFirst;
                top++;
                indexStack[top] = node.leftFirst + 1;
                top++;
            }
        }

        top--;
        node = bvhNodes[indexStack[top]];
    } while(top > -1);

    return _result;
}

bool isInShadow(vec3 lightPos, RaycastResult raycastHit)
{
    Ray shadowRay;
    shadowRay.origin = raycastHit.position + raycastHit.normal * 0.001;
    shadowRay.direction = normalize(lightPos - raycastHit.position);
    
    float lightDistance = length(lightPos - raycastHit.position);
    
    int indexStack[64];
    int top = 0;
    indexStack[top] = 0;
    int hits = 0;

    Node node = bvhNodes[indexStack[top]];
    top++;

    do
    {
        if(IntersectAABB(shadowRay, node.minBound, node.maxBound))
        {
            if(node.triangleCount > 0)
            {
                for(int i = 0; i < node.triangleCount; i++)
                {
                    if(bvhTriangleIndexes[node.leftFirst + i] == raycastHit.triangleIndex)
                    {
                        continue;
                    }

                    RaycastResult shadowHit = triangleIntersect(shadowRay, modelData[bvhTriangleIndexes[node.leftFirst + i]], bvhTriangleIndexes[node.leftFirst + i]);
                    if(shadowHit.hit && shadowHit.distance < lightDistance)
                    {
                        return true;
                    }
                }
            }
            else
            {
                indexStack[top] = node.leftFirst;
                top++;
                indexStack[top] = node.leftFirst + 1;
                top++;
            }
        }

        top--;
        node = bvhNodes[indexStack[top]];
    } while(top > -1);

    return false;
}


vec3 getObjectColor(RaycastResult hit)
{
    //return normalize(vec3(3.0f - hit.distance));
    return vec3(1.0, 0.0, 0.0);
}

vec3 calculateShading(RaycastResult hit, Ray _ray)
{
    if (!hit.hit)
    {
        return vec3(0.1, 0.1, 0.3); // Background color
    }
    
    vec3 objectColor = getObjectColor(hit);

    //return objectColor;

    vec3 lighting = vec3(0.0);

    // Flip normal if it's facing away from the ray (backface)
    vec3 normal = hit.normal;
    if(dot(normal, _ray.direction) > 0.0)
    {
        normal = -normal;
    }

    for(int i = 0; i < numLights; i++)
    {
        // Check if point is in shadow for this light
        if (isInShadow(lights[i].position, hit))
        {
            continue;
        }
        
        // Calculate direction from hit point to light
        vec3 lightDir = normalize(lights[i].position - hit.position);

        // Diffuse component
        float diffuse = max(dot(normal, lightDir), 0.0);
        vec3 diffuseColor = lights[i].color * objectColor * diffuse;

        // Specular component
        vec3 viewDir = normalize(u_camera.position - hit.position);
        vec3 reflectDir = reflect(-lightDir, normal);
        float specular = pow(max(dot(viewDir, reflectDir), 0.0), 32.0);
        vec3 specularColor = lights[i].color * specular;

        // Add attenuation based on distance
        float distance = length(lights[i].position - hit.position);
        float attenuation = 1.0 / (1.0 + 0.1 * distance + 0.01 * distance * distance);
        
        lighting += (diffuseColor + specularColor) * attenuation;
    };
    
    // Add ambient lighting (always present, even in shadows)
    vec3 ambient = objectColor * 0.1;
    
    return ambient + lighting;
}
