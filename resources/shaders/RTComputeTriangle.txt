#version 430 core

layout(local_size_x = 4, local_size_y = 4) in;
layout(rgba32f, binding = 0) uniform image2D framebufferTex;

struct Triangle
{
    int materialIndex;
    vec3 a;
    vec3 b;
    vec3 c;
    vec3 normal;
};

layout(std430, binding = 1) readonly buffer b_trianglesSSBO
{
    Triangle modelData[];
};

uniform vec3 u_modelPosition;

struct Node
{
    vec3 minBound;
    vec3 maxBound;
    int leftIndex;
    int rightIndex;
    int triOffset;
    int triCount;
};

layout(std430, binding = 2) readonly buffer b_bvhNodes
{
    Node bvhNodes[];
};

struct Camera
{
    vec3 position;
    vec3 forward;
    vec3 right;
    vec3 up;
    float fov;
};

uniform Camera u_camera;

uniform vec2 resolution;

struct Light
{
    vec3 color;
    vec3 position;
};

#define MAX_LIGHTS 10
uniform Light lights[MAX_LIGHTS];
uniform int numLights;

struct Ray 
{
    vec3 origin;
    vec3 direction;
};

struct RaycastResult 
{
    bool hit;
    float distance;
    vec3 position;
    vec3 normal;
    int triangleIndex;
    int materialIndex;
};

bool rayNodeIntersect(Ray _ray, Node _node, float _tmax)
{
    vec3 l_invDir = 1.0 / _ray.direction;
    vec3 l_t0 = (_node.minBound - _ray.origin) * l_invDir;
    vec3 l_t1 = (_node.maxBound - _ray.origin) * l_invDir;

    float tmin = 0.0;
    vec3 tmin2 = min(l_t0, l_t1);
    vec3 tmax2 = max(l_t0, l_t1);
    
    tmin = max(tmin, max(tmin2.x, max(tmin2.y, tmin2.z)));
    _tmax = min(_tmax, min(tmax2.x, min(tmax2.y, tmax2.z)));
    
    return tmin <= _tmax;
}

RaycastResult triangleIntersect(Ray _ray, Triangle _tri, int _index)
{
    RaycastResult result;
    result.hit = false;
    result.objectIndex = _index;

    float l_ep = 0.0001;

    vec3 v0 = _tri.a + u_modelPosition;
    vec3 v1 = _tri.b + u_modelPosition; 
    vec3 v2 = _tri.c + u_modelPosition;

    vec3 edge1 = v1 - v0;
    vec3 edge2 = v2 - v0;
    vec3 ray_cross_edge2 = cross(_ray.direction, edge2);
    float det = dot(edge1, ray_cross_edge2);

    // Only check triangles facing the ray
    if(det < l_ep)
    {
        return result;
    }

    float inv_det = 1.0 / det;
    vec3 s = _ray.origin - v0;
    float u = inv_det * dot(s, ray_cross_edge2);

    if(u < 0.0 || u > 1.0)
    {
        return result;
    }

    vec3 s_cross_e1 = cross(s, edge1);
    float v = inv_det * dot(_ray.direction, s_cross_e1);

    if(v < 0.0 || u + v > 1.0)
    {
        return result;
    }

    float t = inv_det * dot(edge2, s_cross_e1);

    if(t > l_ep)
    {
        result.hit = true;
        result.position = _ray.origin + _ray.direction * t;
        result.distance = t;
        result.normal = _tri.normal;
        return result;
    }

    return result;
}

bool isInShadow(vec3 lightPos, RaycastResult raycastHit)
{
    Ray shadowRay;
    shadowRay.origin = raycastHit.position + raycastHit.normal * 0.001;
    shadowRay.direction = normalize(lightPos - raycastHit.position);
    
    float lightDistance = length(lightPos - raycastHit.position);
    
    for (int i = 0; i < modelData.length(); i++)
    {
        if (i == raycastHit.objectIndex)
        {
            continue;
        }
        
        RaycastResult shadowHit = triangleIntersect(shadowRay, modelData[i], i);
        if (shadowHit.hit && shadowHit.distance < lightDistance)
        {
            return true;
        }
    }
    
    return false;
}


vec3 getObjectColor(RaycastResult hit)
{
    return modelData[hit.objectIndex].normal;
}

vec3 calculateShading(RaycastResult hit, Ray ray)
{
    if (!hit.hit)
    {
        return vec3(0.1, 0.1, 0.3); // Background color
    }
    
    vec3 objectColor = getObjectColor(hit);
    vec3 lighting = vec3(0.0);

    for(int i = 0; i < numLights; i++)
    {
        // Check if point is in shadow for this light
        if (isInShadow(lights[i].position, hit))
        {
            continue;
        }
        
        // Calculate direction from hit point to light
        vec3 lightDir = normalize(lights[i].position - hit.position);

        // Diffuse component
        float diffuse = max(dot(hit.normal, lightDir), 0.0);
        vec3 diffuseColor = lights[i].color * objectColor * diffuse;

        // Specular component
        vec3 viewDir = normalize(u_camera.position - hit.position);
        vec3 reflectDir = reflect(-lightDir, hit.normal);
        float specular = pow(max(dot(viewDir, reflectDir), 0.0), 32.0);
        vec3 specularColor = lights[i].color * specular;

        // Add attenuation based on distance
        float distance = length(lights[i].position - hit.position);
        float attenuation = 1.0 / (1.0 + 0.1 * distance + 0.01 * distance * distance);
        
        lighting += (diffuseColor + specularColor) * attenuation;
    };
    
    // Add ambient lighting (always present, even in shadows)
    vec3 ambient = objectColor * 0.1;
    
    return ambient + lighting;
}

void main() 
{
    vec2 uv = vec2(((float(gl_GlobalInvocationID.x) / gl_NumWorkGroups.x) - 0.5) * 2.0,
                   ((float(gl_GlobalInvocationID.y) / gl_NumWorkGroups.y) - 0.5) * 2.0);

    float aspect = gl_NumWorkGroups.x / gl_NumWorkGroups.y;
    vec3 rayDir = normalize(u_camera.forward + 
                           uv.x * u_camera.right * aspect + 
                           uv.y * u_camera.up);
    
    Ray ray;
    ray.origin = u_camera.position;
    ray.direction = rayDir;
    
    RaycastResult closestHit;
    closestHit.hit = false;
    closestHit.distance = 1e10;

    int l_triCount = modelData.length();
    for (int i = 0; i < l_triCount; i++)
    {
        // Skip back facing triangles
        vec3 difference = ray.direction - modelData[i].normal;
        if(length(difference) < 0.1)
        {
            continue;
        }

        RaycastResult hit = triangleIntersect(ray, modelData[i], i);
        if (hit.hit && hit.distance < closestHit.distance)
        {
            closestHit = hit;
        }
    }

    vec3 color = calculateShading(closestHit, ray);
    ivec2 position = ivec2(gl_GlobalInvocationID.xy);
    imageStore(framebufferTex, position, vec4(color, 1.0));
}
