#version 430 core

layout(local_size_x = 4, local_size_y = 4) in;

layout(rgba32f, binding = 0) uniform image2D o_framebufferTex;

layout(rgba32f, binding = 1) uniform image2D intersectPositions;
layout(rgba32f, binding = 2) uniform image2D triangleNormals;
layout(rgba32f, binding = 3) uniform image2D uvCoordinatesAndTexIdx;
layout(rgba32f, binding = 4) uniform image2D matAlbedo;
layout(rgba32f, binding = 5) uniform image2D matProperties;

layout(r32f, binding = 6) uniform image2D shadowBuffer;

#define MAX_TEXTURES 10
layout(binding = 11) uniform sampler2D u_materialTextures[MAX_TEXTURES];

struct Camera { vec3 position; vec3 forward; vec3 right; vec3 up; };
uniform Camera u_camera;

struct Light{ vec3 color; vec3 position; };
#define MAX_LIGHTS 10
uniform Light u_lights[MAX_LIGHTS];
uniform int u_numLights;


void main()
{
    ivec2 l_pixelCoords = ivec2(gl_GlobalInvocationID.xy);
    vec3 l_position = imageLoad(intersectPositions, l_pixelCoords).xyz;

    // if(length(l_position) == 0.0f)
    // {
    //     imageStore(o_framebufferTex, l_pixelCoords, vec4(0.3, 0.1, 0.1, 1.0f));
    //     return;
    // }

    int l_texId = int(imageLoad(intersectPositions, l_pixelCoords).b);
    vec2 l_tevUVs = imageLoad(uvCoordinatesAndTexIdx, l_pixelCoords).rg;
    
    vec3 l_objectColor = texture(u_materialTextures[l_texId], ivec2(l_tevUVs.x, l_tevUVs.y)).rgb;

    vec3 l_lighting = vec3(0.0);

    // For some reason the normal is 99% of the time ALWAYS FUCKING BACKWARDS
    // Flip normal if it's facing away from the ray (backface)
    vec3 l_normal = imageLoad(triangleNormals, l_pixelCoords).rgb;

    for(int i = 0; i < u_numLights; i++)
    {   
        vec3 l_lightDir = normalize(u_lights[i].position - l_position);

        float l_diffuse = max(dot(l_normal, l_lightDir), 0.0);
        vec3 l_diffuseColor = u_lights[i].color * l_objectColor * l_diffuse;

        vec3 l_viewDir = normalize(u_camera.position - l_position);
        vec3 l_reflectDir = reflect(-l_lightDir, l_normal);

        float l_specular = pow(max(dot(l_viewDir, l_reflectDir), 0.0), 32.0);
        vec3 l_specularColor = u_lights[i].color * l_specular;

        // Add attenuation based on distance
        float l_distance = length(u_lights[i].position - l_position);

        // Scales light amount by inverse of distance
        float l_attenuation = 1.0 / (1.0 + 0.1 * l_distance + 0.01 * l_distance * l_distance);
        
        l_lighting += (l_diffuseColor + l_specularColor) * l_attenuation;
    };
    
    // Add ambient lighting (always present, even in shadows)
    vec3 l_ambient = l_objectColor * 0.1;
    
    vec3 colour = (l_ambient + l_lighting) * imageLoad(shadowBuffer, l_pixelCoords).r;

    imageStore(o_framebufferTex, l_pixelCoords, vec4(normalize(l_position), 1.0f));
}
