#version 430 core

layout(local_size_x = 8, local_size_y = 4) in;
layout(rgba32f, binding = 0) uniform image2D o_framebufferTex;

layout(rgba32f, binding = 1) uniform image2D hitLocations;
layout(rgba32f, binding = 2) uniform image2D triangleNormals;

layout(rgba32f, binding = 4) uniform image2D shadowBuffer;
layout(rgba32ui, binding = 5) uniform uimage2D randomBuffer;


struct Triangle
{
    // Vertex positions
    vec3 a;
    vec3 b;
    vec3 c;
    vec3 normal;

    // UV Coordinates for each vertex
    vec2 uvA;
    vec2 uvB;
    vec2 uvC;

    int textureId;
    int materialId;
};

layout(std430, binding = 6) readonly buffer b_trianglesSSBO
{
    Triangle sceneTris[];
};

struct Node
{
    // Laid out like this for bit packing
    vec3 minBound;

    // Holds index of left node (if not leaf node) or first triangle (if leaf node)
    int leftFirst;
    vec3 maxBound;

    // Triangles in node (0 if not leaf node)
    int triangleCount;
};

layout(std430, binding = 7) readonly buffer b_bvhNodes
{
    Node bvhNodes[];
};

// Maps bvhNode.leftFirst indexes to indexes in sceneTris
layout(std430, binding = 8) readonly buffer b_bvhTriangleId
{
    int bvhTriIdxs[];
};

#define MAX_LIGHTS 10
struct Light
{
    vec3 points[4];
    vec3 colour;
    float intensity;
    float radius;
};

uniform Light u_lights[MAX_LIGHTS];
uniform int u_lightCount;

uniform int u_samplesPerFrame;

uniform int u_frameCount;

uniform uvec4 u_initialRandom;

struct Ray  { vec3 origin; vec3 direction; };

struct RaycastResult {
    bool hit;
    float distance;
    vec3 position;
    vec3 normal;
    int triangleId;
};

const float PI = 3.14159265359;

RaycastResult TriangleIntersect(Ray _ray, Triangle _tri, int _index)
{
    RaycastResult result;
    result.hit = false;
    result.triangleId = _index;

    const float l_ep = 0.000001;
    
    vec3 edge1 = _tri.b - _tri.a;
    vec3 edge2 = _tri.c - _tri.a;
    
    vec3 h = cross(_ray.direction, edge2);
    float det = dot(edge1, h);
    
    // If the determinant is negative, the triangle is back-facing.
    if (det < l_ep)
    {
        return result;
    }
    
    float invDet = 1.0 / det;
    vec3 s = _ray.origin - _tri.a;
    float u = invDet * dot(s, h);
    
    if (u < 0.0 || u > 1.0)
    {
        return result;
    }
    
    vec3 q = cross(s, edge1);
    float v = invDet * dot(_ray.direction, q);
    
    if (v < 0.0 || u + v > 1.0)
    {
        return result;
    }
    
    // At this stage we can compute t to find out where
    // the intersection point is on the line
    float t = invDet * dot(edge2, q);
    
    if (t > l_ep) // Ray intersection
    {
        result.hit = true;
        result.position = _ray.origin + _ray.direction * t;
        result.distance = t;
        result.normal = normalize(cross(edge1, edge2));
        return result;
    }
    
    // This means that there is a line intersection but not a ray intersection
    return result;
}

bool IntersectAABB(inout Ray _ray, vec3 _minBound, vec3 _maxBound)
{
    float tmin = 0.0f;
    float tmax = 10000000.0f;

    for (int i = 0; i < 3; i++)
    {
        if (abs(_ray.direction[i]) < 0.0001f)
        {
            if (_ray.origin[i] < _minBound[i] || _ray.origin[i] > _maxBound[i])
                return false;
        }
        else
        {
            float ood = 1.0f / _ray.direction[i];
            float t1 = (_minBound[i] - _ray.origin[i]) * ood;
            float t2 = (_maxBound[i] - _ray.origin[i]) * ood;

            if (t1 > t2)
            {
                float temp = t1;
                t1 = t2;
                t2 = temp;
            }

            tmin = max(tmin, t1);
            tmax = min(tmax, t2);

            if (tmin > tmax)
            {
                return false;
            }
        }
    }

    return true;
}

// Shadow ray tracing (simplified version of TraverseBVH)
bool IsInShadow(vec3 _lightPos, vec3 _position, vec3 _normal, int _idx, inout vec3 _error)
{
    Ray l_shadowRay;
    l_shadowRay.origin = _position + _normal * 0.01;
    l_shadowRay.direction = normalize(_lightPos - _position);
    
    float l_lightDistance = length(_lightPos - _position);
    
    int l_indexStack[64];
    int l_top = 0;
    l_indexStack[l_top] = 0;

    Node l_node = bvhNodes[l_indexStack[l_top]];
    l_top++;

    do
    {
        if(IntersectAABB(l_shadowRay, l_node.minBound, l_node.maxBound))
        {
            _error += vec3(0.02, 0.0, 0.0);
            if(l_node.triangleCount > 0)
            {
                for(int i = 0; i < l_node.triangleCount; i++)
                {
                    _error += vec3(0.0, 0.0, 0.01);
                    int triIdx = bvhTriIdxs[l_node.leftFirst + i];
                    RaycastResult newResult = TriangleIntersect(l_shadowRay, sceneTris[triIdx], triIdx);
                    if(newResult.hit && newResult.distance < l_lightDistance)
                    {
                        _error += vec3(0.0, 0.5, 0.00);
                        return true;
                    }
                }
            }
            else
            {
                l_indexStack[l_top] = l_node.leftFirst;
                l_top++;
                l_indexStack[l_top] = l_node.leftFirst + 1;
                l_top++;
            }
        }

        
        l_top--;
        l_node = bvhNodes[l_indexStack[l_top]];
    } while(l_top > -1);

    return false;
}

void main() 
{
    vec3 l_error = vec3(0.0);
    ivec2 l_pixelCoords = ivec2(gl_GlobalInvocationID.xy);

    uvec4 b_rng = imageLoad(randomBuffer, l_pixelCoords);
    if(length(b_rng) == 0 || u_frameCount == 0)
    {
        uvec4 l_uniqueInit = uvec4(
            gl_GlobalInvocationID.x,
            gl_LocalInvocationID.y,
            gl_GlobalInvocationID.y,
            gl_LocalInvocationID.x
        );

        RandomResult l_res = Random(l_uniqueInit);

        b_rng = u_initialRandom + l_res.state;
    }

    RandomResult l_frameRNG = Random(b_rng);
    
    vec3 b_hitLocation = imageLoad(hitLocations, l_pixelCoords).xyz;
    float b_hitDistance = imageLoad(hitLocations, l_pixelCoords).w;
    int b_triId = int(imageLoad(triangleNormals, l_pixelCoords).w);
    Triangle l_hitTriangle = sceneTris[b_triId];
    
    if (b_hitDistance < 0.001)
    {
        imageStore(shadowBuffer, l_pixelCoords, vec4(1.0)); // No shadow for background
        return;
    }


    vec3 b_previousShadow = imageLoad(shadowBuffer, l_pixelCoords).xyz;
    int b_totalSampleCount = int(imageLoad(shadowBuffer, l_pixelCoords).w);
    
    float l_frameShadow = 0.0f;

    // Sample every light, collect average shadow, then average the sum of average shadows for each sample
    //for(sample)
    //  for(light)
    //      Get random point on light
    //      lightDir = randomPoint - hitLocation
    //      is lightDir obstructed ? shadowInSample += 1
    //  endFor
    //  shadowInSample /= lightCount;
    //  totalShadow += shadowInSample
    //endfor
    // totalShadow /= sampleCount
    // Compare totalShadow to previousFrameShadow
    // Store result

    for(int i = 0; i < u_samplesPerFrame; i++)
    {
        float l_shadowInSample = 0.0f;

        for(int l_lightIdx = 0; l_lightIdx < u_lightCount; l_lightIdx++)
        {
            vec3 l_randomPoint = vec3(0.0f);

            if(u_lights[l_lightIdx].radius > 0.0)
            {
                l_randomPoint = SamplePointOnSphere(l_frameRNG, u_lights[l_lightIdx].points[0], u_lights[l_lightIdx].radius);
            }
            else
            {
                vec3 l_edge1 = normalize(u_lights[l_lightIdx].points[1] - u_lights[l_lightIdx].points[0]);
                vec3 l_edge2 = normalize(u_lights[l_lightIdx].points[2] - u_lights[l_lightIdx].points[0]);
                l_randomPoint = SamplePointOnPlane(l_frameRNG, u_lights[l_lightIdx].points[0], l_edge1, l_edge2);
            }

            if(!IsInShadow(l_randomPoint, b_hitLocation, l_hitTriangle.normal, b_triId, l_error))
            {
                l_shadowInSample++;
            }
        }

        l_shadowInSample /= u_lightCount;
        l_frameShadow += l_shadowInSample;
    }

    l_frameShadow /= u_samplesPerFrame;
    b_totalSampleCount++;

    // Accumulate results over time
    
    float alpha = 1.0 / (b_totalSampleCount + 1.0);
    alpha = clamp(alpha, 0.01, 0.1);

    l_frameShadow = mix(b_previousShadow.x, l_frameShadow, alpha);
    
    imageStore(randomBuffer, l_pixelCoords, l_frameRNG.state);
    imageStore(shadowBuffer, l_pixelCoords, vec4(l_frameShadow, l_frameShadow, l_frameShadow, float(b_totalSampleCount)));
    imageStore(o_framebufferTex, l_pixelCoords, l_frameRNG.state);
}
