#version 430 core

layout(local_size_x = 8, local_size_y = 4) in;
layout(rgba32f, binding = 0) uniform image2D o_framebufferTex;

layout(rgba32f, binding = 1) uniform image2D hitLocations;
layout(rgba32f, binding = 2) uniform image2D triangleNormals;

layout(rgba32f, binding = 4) uniform image2D shadowBuffer;

// Statistical shadows based on this paper
// https://boksajak.github.io/files/RTG1_RayTracedShadows.pdf

// STATISTICAL SHADOW PARAMETERS
uniform float u_searchRadius = 0.1;        // Search radius in world units
uniform float u_minVariance = 0.000001;    // Minimum variance to avoid divide by zero
uniform int u_neighborCount = 8;           // Number of neighboring pixels to sample
uniform float u_bias = 0.001;              // Shadow bias

struct Triangle
   
{
    vec3 a; vec3 b; vec3 c;
    vec3 normal;
    vec2 uvA; vec2 uvB; vec2 uvC;
    int textureId;
    int materialId;
};

layout(std430, binding = 6) readonly buffer b_trianglesSSBO
{
    Triangle sceneTris[];
};

struct Node
{
    vec3 minBound;
    int leftFirst;
    vec3 maxBound;
    int triangleCount;
};

layout(std430, binding = 7) readonly buffer b_bvhNodes
{
    Node bvhNodes[];
};

layout(std430, binding = 8) readonly buffer b_bvhTriangleId
{
    int bvhTriIdxs[];
};

#define MAX_LIGHTS 10
struct Light
{
    vec3 points[4];
    vec3 colour;
    float intensity;
    float radius;
};

uniform Light u_lights[MAX_LIGHTS];
uniform int u_lightCount;
uniform int u_samplesPerFrame;
uniform int u_frameCount;

struct Ray { vec3 origin; vec3 direction; };

struct RaycastResult
{
    bool hit;
    float distance;
    vec3 position;
    vec3 normal;
    int triangleId;
};

const float PI = 3.14159265359;


// Linear Congruential Generator
// Values chosen from https://en.wikipedia.org/wiki/Linear_congruential_generator#Parameters_in_common_use
uint lcg(inout uint _seed)
{
    _seed = (1664525u * _seed + 1013904223u);
    return _seed;
}

// Returns random number between 0 and 1
float RandomFloat(inout uint _seed)
{
    //                        This is the uint number cap
    return float(lcg(_seed)) / 4294967295.0f;
}

vec3 SamplePointOnSphere(inout uint _seed, vec3 center, float radius)
{
    float u = (RandomFloat(_seed) - 0.5f) * 2.0f;
    float v = (RandomFloat(_seed) - 0.5f) * 2.0f;

    // Turn random numbers into cosine distribution
    float theta = 2.0 * PI * u;
    float phi = PI * v;

    vec3 dir = normalize(vec3(
                                cos(theta) * sin(phi),
                                sin(theta) * sin(phi),
                                cos(phi)
                             ));
    return center + dir * radius;
}

vec3 SamplePointOnPlane(inout uint _seed, vec3 _corner, vec3 _edge1, vec3 _edge2)
{
    float u = RandomFloat(_seed);
    float v = RandomFloat(_seed);

    return _corner + u * _edge1 + v * _edge2;
}

// Chebyshev's inequality for variance shadow estimation
float ChebyshevUpperBound(float _moment1, float _moment2, float _t)
{
    // _t is the current depth
    // _moment1 is the mean (M1)
    // _moment2 is the mean of squares (M2)
    
    if (_t <= _moment1)
    {
        // Surface is in front of the mean -> fully lit
        return 1.0;
    }
    
    float variance = max(_moment2 - _moment1 * _moment1, u_minVariance);
    float d = _t - _moment1;
    float p_max = variance / (variance + d * d);
    
    return clamp(p_max, 0.0, 1.0);
}

// Single ray trace for depth value
float TraceShadowRay(vec3 origin, vec3 direction, float maxDistance)
{
    Ray ray;
    ray.origin = origin;
    ray.direction = direction;
    
    float closestHit = maxDistance;
    
    int stack[64];
    int stackPtr = 0;
    stack[stackPtr++] = 0;
    
    while (stackPtr > 0)
    {
        int nodeIdx = stack[--stackPtr];
        Node node = bvhNodes[nodeIdx];
        
        // AABB intersection
        float tmin = 0.0;
        float tmax = maxDistance;
        
        for (int i = 0; i < 3; i++) 
        {
            if (abs(direction[i]) < 0.0001) 
            {
                if (origin[i] < node.minBound[i] || origin[i] > node.maxBound[i]) 
                {
                    tmin = tmax + 1.0;
                    break;
                }
            } 
            else 
            {
                float invDir = 1.0 / direction[i];
                float t1 = (node.minBound[i] - origin[i]) * invDir;
                float t2 = (node.maxBound[i] - origin[i]) * invDir;
                
                if (t1 > t2) 
                {
                    float temp = t1;
                    t1 = t2;
                    t2 = temp;
                }
                
                tmin = max(tmin, t1);
                tmax = min(tmax, t2);
                
                if (tmin > tmax) 
                {
                    tmin = tmax + 1.0;
                    break;
                }
            }
        }
        
        if (tmin <= tmax) 
        {
            if (node.triangleCount > 0) 
            {
                // Leaf node - check triangles
                for (int i = 0; i < node.triangleCount; i++) 
                {
                    int triIdx = bvhTriIdxs[node.leftFirst + i];
                    Triangle tri = sceneTris[triIdx];
                    
                    // Triangle intersect thrown in here for 
                    vec3 edge1 = tri.b - tri.a;
                    vec3 edge2 = tri.c - tri.a;
                    vec3 h = cross(direction, edge2);
                    float det = dot(edge1, h);
                    
                    if (abs(det) < 0.000001)
                        continue;
                        
                    float invDet = 1.0 / det;
                    vec3 s = origin - tri.a;
                    float u = invDet * dot(s, h);
                    
                    if (u < 0.0 || u > 1.0)
                        continue;
                        
                    vec3 q = cross(s, edge1);
                    float v = invDet * dot(direction, q);
                    
                    if (v < 0.0 || u + v > 1.0)
                        continue;
                        
                    float l_t = invDet * dot(edge2, q);
                    
                    if (l_t > 0.0001 && l_t < closestHit) 
                    {
                        closestHit = l_t;
                    }
                }
            } 
            else 
            {
                // Interior node - push children
                stack[stackPtr++] = node.leftFirst;
                stack[stackPtr++] = node.leftFirst + 1;
            }
        }
    }
    
    return closestHit;
}

// Gather neighboring depth moments for statistical estimation
vec2 GatherDepthMoments(vec3 worldPos, vec3 _normal, vec3 _lightDir, float lightDistance, inout uint _rngState) 
{
    vec2 moments = vec2(0.0);
    float weightSum = 0.0;
    
    // Sample random points within search radius
    for (int i = 0; i < u_neighborCount; i++) 
    {
        // Generate random offset within search radius
        vec2 offset = vec2(
            RandomFloat(_rngState) * 2.0 - 1.0,
            RandomFloat(_rngState) * 2.0 - 1.0
        );
        
        if (dot(offset, offset) > 1.0) 
        {
            i--; // Rejection sampling
            continue;
        }
        
        offset *= u_searchRadius / lightDistance;
        
        // Jittered ray direction
        vec3 jitteredDir = normalize(_lightDir + 
                                     vec3(offset.x, offset.y, 0.0) * 
                                     (1.0 - abs(dot(_normal, _lightDir))));
        
        // Trace to get depth
        float depth = TraceShadowRay(worldPos + _normal * u_bias, 
                                    jitteredDir, lightDistance);
        
        // Convert to light space depth (0 at light, 1 at surface)
        float lightSpaceDepth = depth / lightDistance;
        
        // Weight by cosine between normal and light direction
        float weight = max(dot(_normal, jitteredDir), 0.0);
        
        // Exponential moving average for moments
        moments += vec2(lightSpaceDepth, lightSpaceDepth * lightSpaceDepth) * weight;
        weightSum += weight;
    }
    
    if (weightSum > 0.0) 
    {
        moments /= weightSum;
    }
    
    return moments;
}

void main() 
{
    ivec2 l_pixelCoords = ivec2(gl_GlobalInvocationID.xy);

    uint l_rngState = uint(l_pixelCoords.x * 1973 + l_pixelCoords.y * 9277 + u_frameCount * 26699);

    // Load hit data
    vec4 b_hitData = imageLoad(hitLocations, l_pixelCoords);
    vec3 l_hitLocation = b_hitData.xyz;
    float l_hitDistance = b_hitData.w;
    
    vec4 b_normalData = imageLoad(triangleNormals, l_pixelCoords);
    vec3 l_normal = b_normalData.xyz;
    int l_triId = int(b_normalData.w);
    
    if (l_hitDistance < 0.001) 
    {
        imageStore(shadowBuffer, l_pixelCoords, vec4(1.0));
        return;
    }
    
    // Load previous shadow result and sample count
    vec4 b_prevShadowData = imageLoad(shadowBuffer, l_pixelCoords);
    vec3 l_prevShadow = b_prevShadowData.xyz;
    int l_totalSamples = int(b_prevShadowData.w);
    
    float l_shadowInFrame = 0.0;
    int l_validLights = 0;
    
    // Process each light
    for (int i = 0; i < u_lightCount; i++) 
    {
        Light l_light = u_lights[i];
        vec3 l_lightPos = l_light.points[0];
        
        // Calculate light direction and distance
        vec3 l_lightDir = l_lightPos - l_hitLocation;
        float l_lightDistance = length(l_lightDir);
        l_lightDir = normalize(l_lightDir);
        
        // Skip if light is behind surface
        float NdotL = dot(l_normal, l_lightDir);
        if (NdotL <= 0.0) 
        {
            continue;
        }
        
        // Gather statistical depth moments
        vec2 l_moments = GatherDepthMoments(l_hitLocation, l_normal, l_lightDir, l_lightDistance, l_rngState);
        
        // Current depth in light space (0 = at light, 1 = at surface)
        float l_currentDepth = 1.0; // We're at the surface
        
        // Apply Chebyshev's inequality
        float l_visibility = ChebyshevUpperBound(l_moments.x, l_moments.y, l_currentDepth);
        
        // Apply distance attenuation (optional)
        float l_attenuation = 1.0 / (1.0 + 0.1 * l_lightDistance * l_lightDistance);
        
        l_shadowInFrame += l_visibility * l_attenuation * l_light.intensity * NdotL;
        l_validLights++;
    }
    
    // Average over lights
    if (l_validLights > 0) 
    {
        l_shadowInFrame /= float(l_validLights);
    } else
    {
        l_shadowInFrame = 0.0;
    }
    
    // Temporal accumulation
    l_totalSamples++;
    float alpha = 1.0 / float(l_totalSamples);
    alpha = clamp(alpha, 0.01, 0.1);
    
    l_shadowInFrame = mix(l_prevShadow.x, l_shadowInFrame, alpha);
    
    // Store results
    imageStore(shadowBuffer, l_pixelCoords, vec4(vec3(l_shadowInFrame), float(l_totalSamples)));
}
