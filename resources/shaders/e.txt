//Returns distance to cube or 1e30f if miss
float IntersectAABB(in Ray _ray, vec3 _minBound, vec3 _maxBound)
{
    float tx1 = (_minBound.x - _ray.origin.x) * _ray.recipDir.x, tx2 = (_maxBound.x - _ray.origin.x) * _ray.recipDir.x;
    float tmin = min( tx1, tx2 ), tmax = max( tx1, tx2 );
    float ty1 = (_minBound.y - _ray.origin.y) * _ray.recipDir.y, ty2 = (_maxBound.y - _ray.origin.y) * _ray.recipDir.y;
    tmin = max( tmin, min( ty1, ty2 ) ), tmax = min( tmax, max( ty1, ty2 ) );
    float tz1 = (_minBound.z - _ray.origin.z) * _ray.recipDir.z, tz2 = (_maxBound.z - _ray.origin.z) * _ray.recipDir.z;
    tmin = max( tmin, min( tz1, tz2 ) ), tmax = min( tmax, max( tz1, tz2 ) );
    if (tmax >= tmin && tmin < _ray.distance && tmax > 0) return tmin; else return 1e30f;
}


RaycastResult TraverseBVH(inout RaycastResult _result, inout Ray _ray)
{
    int l_indexStack[64];
    int l_top = 0;
    l_indexStack[l_top] = 0;

    Node l_node = bvhNodes[l_indexStack[l_top]];
    l_top++;

    while(true)
    {
        if(l_node.triCount > 0)
        {
            for(int i = 0; i < l_node.triCount; i++)
            {
                RaycastResult newResult = triangleIntersect(_ray, triangles[bvhTriIdx[int(l_node.leftFirst) + i]], bvhTriIdx[int(l_node.leftFirst) + i]);
                if(newResult.hit && newResult.distance < _result.distance)
                {
                    _result = newResult;
                }
            }

            if(l_top == 0)
            {
                break;
            }
            
            l_top--;
            l_node = bvhNodes[l_top];
        }

        Node l_child1 = bvhNodes[int(l_node.leftFirst)];
        Node l_child2 = bvhNodes[int(l_node.leftFirst + 1)];

        float l_dist1 = IntersectAABB(_ray, l_child1.minBound, l_child1.maxBound);
        float l_dist2 = IntersectAABB(_ray, l_child2.minBound, l_child2.maxBound);

        if(l_dist1 > l_dist2)
        {
            float tempF = l_dist1;
            l_dist1 = l_dist2;
            l_dist2 = tempF;

            Node tempN = l_child1;
            l_child1 = l_child2;
            l_child2 = tempN;
        }

        if(l_dist1> 1e30f)
        {
            if(l_top == 0)
            {
                break;
            }


            l_top--;
            l_node = bvhNodes[l_indexStack[l_top]];
        }
        else
        {
            if(l_dist2 != 1e30f)
            {
                l_top++;
                l_indexStack[l_top] = int(l_node.leftFirst) + 1;
            }
            l_node = l_child1;
        }
    }

    return _result;
}
