542inline void ImBitArraySetBitRange(ImU32* arr, int n, int n2) // Works on range [n..n2)
543{
544 n2--;
545 while (n <= n2)
546 {
547 int a_mod = (n & 31);
548 int b_mod = (n2 > (n | 31) ? 31 : (n2 & 31)) + 1;
549 ImU32 mask = (ImU32)(((ImU64)1 << b_mod) - 1) & ~(ImU32)(((ImU64)1 << a_mod) - 1);
550 arr[n >> 5] |= mask;
551 n = (n + 32) & ~31;
552 }

699// Estimation of number of circle segment based on error is derived using method described in https://stackoverflow.com/a/2244088/15194693
700// Number of segments (N) is calculated using equation:
701// N = ceil ( pi / acos(1 - error / r) ) where r > 0, error <= r
702// Our equation is significantly simpler that one in the post thanks for choosing segment that is
703// perpendicular to X axis. Follow steps in the article from this starting condition and you will
704// will get this result.
705//
706// Rendering circles with an odd number of segments, while mathematically correct will produce
707// asymmetrical results on the raster grid. Therefore we're rounding N to next even number (7->8, 8->8, 9->10 etc.)
708#define IM_ROUNDUP_TO_EVEN(_V) ((((_V) + 1) / 2) * 2)
709#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MIN 4
710#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MAX 512
711#define IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_CALC(_RAD,_MAXERROR) ImClamp(IM_ROUNDUP_TO_EVEN((int)ImCeil(IM_PI / ImAcos(1 - ImMin((_MAXERROR), (_RAD)) / (_RAD)))), IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MIN, IM_DRAWLIST_CIRCLE_AUTO_SEGMENT_MAX)
