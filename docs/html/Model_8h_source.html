79inline Model::Model(const std::string& _path) :
80 m_dirty(false)
81 , m_position()
82 , m_ssboId(0)
83{
84 std::vector<glm::vec3> L_positions;
85 std::vector<glm::vec2> L_tcs;
86 std::vector<glm::vec3> L_normals;
87 std::string L_currentline;
88
89 std::ifstream file(_path.c_str());
90
91 if(!file.is_open())
92 {
93 throw std::runtime_error("Failed to open model [" + _path + "]");
94 }
95
96 while(!file.eof())
97 {
98 std::getline(file, L_currentline);
99 if(L_currentline.length() < 1) continue;
100
101 std::vector<std::string> L_tokens;
102 SplitStringWhitespace(L_currentline, L_tokens);
103 if(L_tokens.size() < 1) continue;
104
105 if(L_tokens.at(0) == "v" && L_tokens.size() >= 4)
106 {
107 glm::vec3 p(atof(L_tokens.at(1).c_str()),
108 atof(L_tokens.at(2).c_str()),
109 atof(L_tokens.at(3).c_str()));
110
111 L_positions.push_back(p);
112 }
113 else if(L_tokens.at(0) == "vt" && L_tokens.size() >= 3)
114 {
115 glm::vec2 tc(atof(L_tokens.at(1).c_str()),
116 1.0f - atof(L_tokens.at(2).c_str()));
117
118 L_tcs.push_back(tc);
119 }
120 else if(L_tokens.at(0) == "vn" && L_tokens.size() >= 4)
121 {
122 glm::vec3 n(atof(L_tokens.at(1).c_str()),
123 atof(L_tokens.at(2).c_str()),
124 atof(L_tokens.at(3).c_str()));
125
126 L_normals.push_back(n);
127 }
128 else if(L_tokens.at(0) == "f" && L_tokens.size() >= 4)
129 {
130 Face f;
131 std::vector<std::string> sub;
132 SplitString(L_tokens.at(1), '/', sub);
133 if(sub.size() >= 1) f.a.position = L_positions.at(atoi(sub.at(0).c_str()) - 1);
134 if(sub.size() >= 2) f.a.texcoord = L_tcs.at(atoi(sub.at(1).c_str()) - 1);
135 if(sub.size() >= 3) f.a.normal = L_normals.at(atoi(sub.at(2).c_str()) - 1);
136
137 for(size_t ti = 2; ti + 1 < L_tokens.size(); ti++)
138 {
139 SplitString(L_tokens.at(ti), '/', sub);
140 if(sub.size() >= 1) f.b.position = L_positions.at(atoi(sub.at(0).c_str()) - 1);
141 if(sub.size() >= 2) f.b.texcoord = L_tcs.at(atoi(sub.at(1).c_str()) - 1);
142 if(sub.size() >= 3) f.b.normal = L_normals.at(atoi(sub.at(2).c_str()) - 1);
143
144 SplitString(L_tokens.at(ti + 1), '/', sub);
145 if(sub.size() >= 1) f.c.position = L_positions.at(atoi(sub.at(0).c_str()) - 1);
146 if(sub.size() >= 2) f.c.texcoord = L_tcs.at(atoi(sub.at(1).c_str()) - 1);
147 if(sub.size() >= 3) f.c.normal = L_normals.at(atoi(sub.at(2).c_str()) - 1);
148
149 f.CalculateNormal();
150 m_faces.push_back(f);
151 m_dirty = true;
152 }
153 }
154 }

179inline void Model::SplitStringWhitespace(const std::string& _input,
180 std::vector<std::string>& _output)
181{
182 std::string curr;
183
184 _output.clear();
185
186 for(size_t i = 0; i < _input.length(); i++)
187 {
188 if(_input.at(i) == ' ' ||
189 _input.at(i) == '\r' ||
190 _input.at(i) == '\n' ||
191 _input.at(i) == '\t')
192 {
193 if(curr.length() > 0)
194 {
195 _output.push_back(curr);
196 curr = "";
197 }
198 }
199 else
200 {
201 curr += _input.at(i);
202 }
203 }
204
205 if(curr.length() > 0)
206 {
207 _output.push_back(curr);
